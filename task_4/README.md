## Табличные пространства
```sql
DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_tablespace WHERE spcname = 'ts_core_data') THEN
    EXECUTE 'CREATE TABLESPACE ts_core_data LOCATION ''/var/lib/postgresql/tablespaces/ts_core_data''';
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_tablespace WHERE spcname = 'ts_ops_data') THEN
    EXECUTE 'CREATE TABLESPACE ts_ops_data LOCATION ''/var/lib/postgresql/tablespaces/ts_ops_data''';
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_tablespace WHERE spcname = 'ts_audit_data') THEN
    EXECUTE 'CREATE TABLESPACE ts_audit_data LOCATION ''/var/lib/postgresql/tablespaces/ts_audit_data''';
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_tablespace WHERE spcname = 'ts_idx') THEN
    EXECUTE 'CREATE TABLESPACE ts_idx LOCATION ''/var/lib/postgresql/tablespaces/ts_idx''';
  END IF;
END$$;
```

## Роли
```sql
DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'app_owner') THEN
    CREATE ROLE app_owner LOGIN PASSWORD 'change_me_owner'
      NOSUPERUSER NOCREATEDB NOCREATEROLE NOINHERIT;
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'app_rw') THEN
    CREATE ROLE app_rw LOGIN PASSWORD 'change_me_rw'
      NOSUPERUSER NOCREATEDB NOCREATEROLE NOINHERIT;
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'app_ro') THEN
    CREATE ROLE app_ro LOGIN PASSWORD 'change_me_ro'
      NOSUPERUSER NOCREATEDB NOCREATEROLE NOINHERIT;
  END IF;
```
## Схема данных
```sql

CREATE SCHEMA IF NOT EXISTS ref   AUTHORIZATION app_owner;  
CREATE SCHEMA IF NOT EXISTS core  AUTHORIZATION app_owner;  
CREATE SCHEMA IF NOT EXISTS wh    AUTHORIZATION app_owner;  
CREATE SCHEMA IF NOT EXISTS ops   AUTHORIZATION app_owner;  
CREATE SCHEMA IF NOT EXISTS audit AUTHORIZATION app_owner;  

GRANT USAGE ON SCHEMA ref, core, wh, ops, audit TO app_rw, app_ro;


ALTER DEFAULT PRIVILEGES FOR ROLE app_owner IN SCHEMA ref, core, wh, ops, audit
  GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO app_rw;

ALTER DEFAULT PRIVILEGES FOR ROLE app_owner IN SCHEMA ref, core, wh, ops, audit
  GRANT USAGE, SELECT, UPDATE ON SEQUENCES TO app_rw;


ALTER DEFAULT PRIVILEGES FOR ROLE app_owner IN SCHEMA ref, core, wh, ops, audit
  GRANT SELECT ON TABLES TO app_ro;

ALTER DEFAULT PRIVILEGES FOR ROLE app_owner IN SCHEMA ref, core, wh, ops, audit
  GRANT USAGE, SELECT ON SEQUENCES TO app_ro;

ALTER ROLE app_rw SET search_path = ops, core, wh, ref, audit, public;
ALTER ROLE app_ro SET search_path = ops, core, wh, ref, audit, public;
```

## Таблицы своего проекта



**warehouses/zones/cells**
```sql
CREATE TABLE IF NOT EXISTS wh.warehouses (
  id       integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name     varchar NOT NULL,
  address  varchar
) TABLESPACE ts_core_data;

CREATE TABLE IF NOT EXISTS wh.zones (
  id           integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  warehouse_id integer NOT NULL REFERENCES wh.warehouses(id) ON DELETE RESTRICT,
  code         varchar NOT NULL,
  name         varchar,
  UNIQUE (warehouse_id, code)
) TABLESPACE ts_core_data;

CREATE TABLE IF NOT EXISTS wh.cells (
  id      integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  zone_id integer NOT NULL REFERENCES wh.zones(id) ON DELETE RESTRICT,
  code    varchar NOT NULL,
  UNIQUE (zone_id, code)
) TABLESPACE ts_core_data;
```

**clients, users, products**
```sql

CREATE TABLE IF NOT EXISTS core.clients (
  id   integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name varchar NOT NULL,
  inn  varchar
) TABLESPACE ts_core_data;

CREATE TABLE IF NOT EXISTS core.users (
  id         integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  login      varchar NOT NULL UNIQUE,
  is_active  boolean NOT NULL DEFAULT true,
  created_at timestamp NOT NULL DEFAULT now()
) TABLESPACE ts_core_data;

CREATE TABLE IF NOT EXISTS core.user_clients (
  user_id   integer NOT NULL REFERENCES core.users(id)   ON DELETE RESTRICT,
  client_id integer NOT NULL REFERENCES core.clients(id) ON DELETE RESTRICT,
  role      varchar NOT NULL,
  PRIMARY KEY (user_id, client_id)
) TABLESPACE ts_core_data;

CREATE TABLE IF NOT EXISTS core.products (
  id        integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  client_id integer NOT NULL REFERENCES core.clients(id) ON DELETE RESTRICT,
  sku       varchar NOT NULL,
  name      varchar NOT NULL,
  gtin      varchar,
  unit      varchar NOT NULL,
  UNIQUE (client_id, sku)
) TABLESPACE ts_core_data;
```

**marking_codes, packages, packages_history, documents**
```sql

CREATE TABLE IF NOT EXISTS ops.marking_codes (
  id         bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  client_id  integer NOT NULL REFERENCES core.clients(id) ON DELETE RESTRICT,
  product_id integer NOT NULL REFERENCES core.products(id) ON DELETE RESTRICT,
  fullcode   varchar NOT NULL,
  searchcode varchar NOT NULL,
  cell_id    integer REFERENCES wh.cells(id) ON DELETE RESTRICT,
  status_id  integer NOT NULL REFERENCES ref.marking_code_statuses(id) ON DELETE RESTRICT,
  created_at timestamp NOT NULL DEFAULT now(),
  UNIQUE (client_id, fullcode)
) TABLESPACE ts_ops_data;

CREATE TABLE IF NOT EXISTS ops.packages (
  id              integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  client_id       integer NOT NULL REFERENCES core.clients(id) ON DELETE RESTRICT,
  package_type_id integer NOT NULL REFERENCES ref.package_types(id) ON DELETE RESTRICT,
  fullcode        varchar NOT NULL,
  searchcode      varchar,
  cell_id         integer NOT NULL REFERENCES wh.cells(id) ON DELETE RESTRICT,
  status_id       integer NOT NULL REFERENCES ref.package_statuses(id) ON DELETE RESTRICT,
  created_at      timestamp NOT NULL DEFAULT now(),
  UNIQUE (client_id, fullcode)
) TABLESPACE ts_ops_data;

CREATE TABLE IF NOT EXISTS ops.packages_history (
  id                bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  client_id         integer NOT NULL REFERENCES core.clients(id) ON DELETE RESTRICT,
  parent_package_id integer NOT NULL REFERENCES ops.packages(id) ON DELETE RESTRICT,
  marking_code_id   bigint REFERENCES ops.marking_codes(id) ON DELETE RESTRICT,
  child_package_id  integer REFERENCES ops.packages(id) ON DELETE RESTRICT,
  added_at          timestamp NOT NULL DEFAULT now(),
  removed_at        timestamp
) TABLESPACE ts_ops_data;

CREATE TABLE IF NOT EXISTS ops.documents (
  id           integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  client_id    integer NOT NULL REFERENCES core.clients(id) ON DELETE RESTRICT,
  doc_type_id  integer NOT NULL REFERENCES ref.doc_types(id) ON DELETE RESTRICT,
  status_id    integer NOT NULL REFERENCES ref.document_statuses(id) ON DELETE RESTRICT,
  created_by   integer NOT NULL REFERENCES core.users(id) ON DELETE RESTRICT,
  created_at   timestamp NOT NULL DEFAULT now(),
  copmleted_ad timestamp
) TABLESPACE ts_ops_data;
```

**events**

```sql
CREATE TABLE IF NOT EXISTS audit.events (
  id            bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  client_id     integer NOT NULL REFERENCES core.clients(id) ON DELETE RESTRICT,
  document_id   integer REFERENCES ops.documents(id) ON DELETE SET NULL,
  event_type_id integer NOT NULL REFERENCES ref.event_types(id) ON DELETE RESTRICT,
  user_id       integer NOT NULL REFERENCES core.users(id) ON DELETE RESTRICT,
  created_at    timestamp NOT NULL DEFAULT now(),
  data          jsonb
) TABLESPACE ts_audit_data;
```
